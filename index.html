const DB_BASE = "https://voice-system-c7785-default-rtdb.firebaseio.com"; const VOICE_SERVER = "wss://voice-server-production-4376.up.railway.app"; let currentUser = null; let ws = null; let peer = null; let localStream = null; let callTimerInterval = null; let callDuration = 0; let currentTarget = null; const rtcConfig = { iceServers: [ { urls: "stun:stun.l.google.com:19302" }, { urls: "stun:stun1.l.google.com:19302" } ] }; window.onload = function() { const session = localStorage.getItem("kurdtell_active_client"); if (session) { currentUser = JSON.parse(session); launchClient(); } }; async function handleAuth() { const phone = document.getElementById("inputPhone").value.trim(); const pass = document.getElementById("inputPass").value.trim(); const alertBox = document.getElementById("authAlert"); const btn = document.getElementById("btnLogin"); alertBox.style.display = "none"; if (!phone || !pass) { alertBox.innerText = "تکایە ژمارەی هێڵ و وشەی نهێنی بنووسە!"; alertBox.style.display = "block"; return; } btn.innerText = "پشکنین..."; btn.disabled = true; try { let res = await fetch(`${DB_BASE}/users/${phone}.json`); let data = await res.json(); if (!data) { let resClients = await fetch(`${DB_BASE}/clients/${phone}.json`); data = await resClients.json(); } if (data) { const remotePass = data.password || data.pass; if (String(remotePass).trim() === String(pass).trim()) { currentUser = { phone: String(phone).trim(), name: data.fullName || data.name || "کڕیار", days: data.daysRemaining || "--" }; localStorage.setItem("kurdtell_active_client", JSON.stringify(currentUser)); launchClient(); } else { alertBox.innerText = "وشەی نهێنی هەڵەیە!"; alertBox.style.display = "block"; } } else { alertBox.innerText = "ئەم هێڵە بوونی نییە لە سیستەم!"; alertBox.style.display = "block"; } } catch (err) { alertBox.innerText = "هەڵەی داتابەیس: " + err.message; alertBox.style.display = "block"; } finally { btn.innerText = "چوونەژوورەوە 📲"; btn.disabled = false; } } function launchClient() { document.getElementById("authBox").style.display = "none"; document.getElementById("clientApp").style.display = "block"; document.getElementById("clientName").innerText = currentUser.name; document.getElementById("clientPhone").innerText = "هێڵ: " + currentUser.phone; document.getElementById("planDays").innerText = currentUser.days + " ڕۆژ ماوە"; initVoiceSocket(); } function handleLogout() { if (ws) ws.close(); localStorage.removeItem("kurdtell_active_client"); location.reload(); } function initVoiceSocket() { const srv = document.getElementById("srvStatus"); srv.innerText = "دەبەسترێت..."; srv.className = "status-offline"; try { ws = new WebSocket(VOICE_SERVER); ws.onopen = () => { srv.innerText = "چالاکە 🟢"; srv.className = "status-online"; // تۆمارکردنی ژمارە لە سێرڤەر ws.send(JSON.stringify({ type: "register", phone: String(currentUser.phone).trim() })); }; ws.onmessage = async (e) => { const msg = JSON.parse(e.data); // کاتێک زەنگ لە کەسی ترەوە دێت if (msg.type === "call" || msg.type === "incoming") { currentTarget = msg.from; const accept = confirm(`📞 پەیوەندی هاتووە لە هێڵی [${msg.from}]\nوەڵام دەدەیتەوە؟`); if (accept) { handleIncomingCall(msg.from, msg.offer); } else { ws.send(JSON.stringify({ type: "reject", to: msg.from })); } } else if (msg.type === "answer") { if (peer) { await peer.setRemoteDescription(new RTCSessionDescription(msg.answer)); onCallConnected(); } } else if (msg.type === "ice") { if (peer && msg.candidate) { try { await peer.addIceCandidate(new RTCIceCandidate(msg.candidate)); } catch (err) {} } } else if (msg.type === "hangup" || msg.type === "reject") { teardownCallUI(); alert("پەیوەندی کۆتایی پێهات یان بەرامبەر ڕەتیکردەوە."); } }; ws.onclose = () => { srv.innerText = "پچڕاوە 🔴"; srv.className = "status-offline"; setTimeout(initVoiceSocket, 3000); }; } catch (err) { srv.innerText = "کێشەی سێرڤەر"; } } function pressKey(num) { const disp = document.getElementById("dialDisplay"); if (disp.innerText.length < 6) disp.innerText += num; } function clearDial() { document.getElementById("dialDisplay").innerText = ""; } function backspaceDial() { const disp = document.getElementById("dialDisplay"); disp.innerText = disp.innerText.slice(0, -1); } async function startVoiceCall() { const target = document.getElementById("dialDisplay").innerText.trim(); if (!target) return alert("تکایە ژمارەی هێڵی بەرامبەر لێبدە!"); if (target === currentUser.phone) return alert("ناتوانیت پەیوەندی بە هێڵی خۆتەوە بکەیت!"); currentTarget = target; openCallUI(target, "زەنگ لێدەدات..."); try { localStream = await navigator.mediaDevices.getUserMedia({ audio: true, video: false }); peer = new RTCPeerConnection(rtcConfig); localStream.getTracks().forEach(track => peer.addTrack(track, localStream)); peer.ontrack = (evt) => { const audio = document.getElementById("remoteVoice"); audio.srcObject = evt.streams[0]; audio.play().catch(e => console.log(e)); }; peer.onicecandidate = (evt) => { if (evt.candidate && ws && ws.readyState === WebSocket.OPEN) { ws.send(JSON.stringify({ type: "ice", to: target, candidate: evt.candidate })); } }; const offer = await peer.createOffer(); await peer.setLocalDescription(offer); // ناردنی بانگهێشتی پەیوەندی بۆ سێرڤەر ws.send(JSON.stringify({ type: "call", from: String(currentUser.phone).trim(), to: String(target).trim(), offer: offer })); } catch (err) { alert("ڕێگەپێدانی مایکرۆفۆن پێویستە بۆ پەیوەندی: " + err.message); teardownCallUI(); } } async function handleIncomingCall(from, offer) { openCallUI(from, "پەیوەست دەبێت..."); try { localStream = await navigator.mediaDevices.getUserMedia({ audio: true, video: false }); peer = new RTCPeerConnection(rtcConfig); localStream.getTracks().forEach(track => peer.addTrack(track, localStream)); peer.ontrack = (evt) => { const audio = document.getElementById("remoteVoice"); audio.srcObject = evt.streams[0]; audio.play().catch(e => console.log(e)); }; peer.onicecandidate = (evt) => { if (evt.candidate && ws && ws.readyState === WebSocket.OPEN) { ws.send(JSON.stringify({ type: "ice", to: from, candidate: evt.candidate })); } }; await peer.setRemoteDescription(new RTCSessionDescription(offer)); const answer = await peer.createAnswer(); await peer.setLocalDescription(answer); ws.send(JSON.stringify({ type: "answer", to: from, answer: answer })); onCallConnected(); } catch (err) { alert("کێشە لە قبوڵکردنی دەنگ: " + err.message); teardownCallUI(); } } function openCallUI(target, state) { document.getElementById("modalTarget").innerText = target; document.getElementById("modalState").innerText = state; document.getElementById("modalTimer").style.display = "none"; document.getElementById("callModal").style.display = "flex"; } function onCallConnected() { document.getElementById("modalState").innerText = "قسەکردن چالاکە 🟢"; const timerElem = document.getElementById("modalTimer"); timerElem.style.display = "block"; callDuration = 0; clearInterval(callTimerInterval); callTimerInterval = setInterval(() => { callDuration++; const mins = String(Math.floor(callDuration / 60)).padStart(2, "0"); const secs = String(callDuration % 60).padStart(2, "0"); timerElem.innerText = `${mins}:${secs}`; }, 1000); } function endVoiceCall() { if (ws && currentTarget && ws.readyState === WebSocket.OPEN) { ws.send(JSON.stringify({ type: "hangup", to: currentTarget })); } teardownCallUI(); } function teardownCallUI() { clearInterval(callTimerInterval); if (peer) { peer.close(); peer = null; } if (localStream) { localStream.getTracks().forEach(t => t.stop()); localStream = null; } currentTarget = null; document.getElementById("callModal").style.display = "none"; }
